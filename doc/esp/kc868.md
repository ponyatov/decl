```decl
hw kc868: esp32s3 {
    board   = 'KC868-A16'
    vendor  = 'KinCony'
    
    // power spec
    vcc     = 12..24<V>      // DC power supply
    current = 500<mA>        // typical consumption
    
    // memory
    flash = 4<Mb>            // ESP-WROOM-32
    
    features = [
	    wifi; eth;           // wireless + wired
	    rs485;               // MODBUS enabled
	    gpio;
	    relay;               // 16 relay outputs
	    dinput               // 16 digital inputs
	]
	
	eth = w5500
	rtc = ds3231
	
	pin = {
	
		// I2C for PCF8574 expanders (16 inputs + 16 outputs)
        i2c_sda = gpio[4]
        i2c_scl = gpio[5]
        
        // RS485 UART
        rs485_tx = gpio[17]
        rs485_rx = gpio[16]
        rs485_en = gpio[21]
        
        // Boot/control pins
        boot   = gpio[0]      // Download button
        
		// relay # to gpio[n] mapping table (index from 1)
		relay:gpio[1..16] = [
			32,33,25,26,27,14,12,13  // 1..8
			15, 2, 0, 4,16,17, 5,18  // 9..16
		]
		
		// digital input to gpio[n] mapping table
		dinput:gpio[1..16] = [
		]
	}
```
