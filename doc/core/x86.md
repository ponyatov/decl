## generate x86 cross dirs & files

```sh
mkdir -p hw/{pc,qemu386}/{inc,src}
mkdir -p cpu/{i5,i686,i486}/{inc,src}
mkdir -p arch/{x86,i386,x86_64}/{inc,src}
mkdir -p os/{linux,win32,win64,bare,rtos}/{inc,src}
```
```sh
touch hw/{pc,qemu386}/{inc,src}/.gitignore
touch cpu/{i5,i686,i486}/{inc,src}/.gitignore
touch arch/{x86,i386,x86_64}/{inc,src}/.gitignore
touch os/{linux,win32,win64,bare,rtos}/{inc,src}/.gitignore
```

```sh
for hw in pc qemu386; do
    touch hw/$hw/$hw.linux
    touch hw/$hw/$hw.uclibc
    touch hw/$hw/inc/$hw.hpp
    touch hw/$hw/src/$hw.cpp
done
```

```sh
for cpu in i5 i686 i486; do
    touch cpu/$cpu/$cpu.linux
    touch cpu/$cpu/$cpu.uclibc
    touch cpu/$cpu/inc/$cpu.hpp
    touch cpu/$cpu/src/$cpu.cpp
done
```

```sh
for arch in x86 i386 x86_64; do
    touch arch/$arch/$arch.linux
    touch arch/$arch/$arch.uclibc
    touch arch/$arch/inc/$arch.hpp
    touch arch/$arch/src/$arch.cpp
done
```
